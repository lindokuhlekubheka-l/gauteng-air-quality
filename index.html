// Group 5 Project - Topic 5: Multi-Pollutant Hotspot Identification and Spatiotemporal Trend Analysis in Gauteng (2018–2025)
// Group Members: Kubheka L.B, Thobejane KI, Mukuru M, Matshete R, Malahleha MS

// Load custom Gauteng boundary asset
var gauteng = ee.FeatureCollection("projects/casenova-499112/assets/Gauteng");
var StudyArea = gauteng.geometry();

Map.centerObject(gauteng, 8);
Map.addLayer(gauteng.style({fillColor: '00000000', color: 'black', width: 2}), {}, 'Gauteng Boundary');

var pal = ['white', 'blue', 'purple', 'red', 'black'];

// MULTI-POLLUTANT PROCESSING (CO, HCHO, SO2, NO2)
var co = ee.ImageCollection('COPERNICUS/S5P/NRTI/L3_CO')
         .filterBounds(StudyArea)
         .filterDate('2018-06-28', '2025-12-31')
         .select('CO_column_number_density')
         .mean().clip(StudyArea);

Map.addLayer(co, {min: 0, max: 0.04, palette: ['white', 'orange', 'red', 'black']}, 'CO Hotspot - Burning', true);

var hcho = ee.ImageCollection('COPERNICUS/S5P/NRTI/L3_HCHO')
         .filterBounds(StudyArea)
         .filterDate('2018-12-01', '2025-12-31')
         .select('tropospheric_HCHO_column_number_density')
         .mean().clip(StudyArea);

Map.addLayer(hcho, {min: 0, max: 0.0002, palette: pal}, 'HCHO Hotspot', true);

var so2 = ee.ImageCollection('COPERNICUS/S5P/NRTI/L3_SO2')
         .filterBounds(StudyArea)
         .filterDate('2018-07-10', '2025-12-31')
         .select('SO2_column_number_density')
         .mean().clip(StudyArea);

Map.addLayer(so2, {min: 0, max: 0.0003, palette: pal}, 'SO2 Hotspot', true);

var no2 = ee.ImageCollection('COPERNICUS/S5P/NRTI/L3_NO2')
         .filterBounds(StudyArea)
         .filterDate('2024-01-01', '2024-12-31')
         .select('NO2_column_number_density')
         .mean().clip(StudyArea);

Map.addLayer(no2, {min: 0, max: 0.0002, palette: pal}, 'NO2 Hotspot', true);

// EXPORTS
Export.image.toDrive({ image: no2, description: 'Gauteng_NO2_2018_2025', folder: 'GEE_Gauteng_AirQuality', region: StudyArea, scale: 1000, crs: 'EPSG:4326', maxPixels: 1e9 });
